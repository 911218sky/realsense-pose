{{- else if contains "NodePort" .Values.service.type }}
📡 Access the application via NodePort:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "realsense-pose.fullname" . }}-nginx)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "ClusterIP" .Values.service.type }}

  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "realsense-pose.fullname" . }}-nginx {{ .Values.service.port }}:{{ .Values.service.port }}
  Then visit: http://localhost:{{ .Values.service.port }}
{{- end }}
